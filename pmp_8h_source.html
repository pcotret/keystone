71#define PMP_SET(n, g, addr, pmpc) \
72{ uintptr_t oldcfg = csr_read(pmpcfg##g); \
73 pmpc |= (oldcfg & ~((uintptr_t)0xff << (uintptr_t)8*(n%PMP_PER_GROUP))); \
74 asm volatile ("la t0, 1f\n\t" \
75 "csrrw t0, mtvec, t0\n\t" \
76 "csrw pmpaddr"#n", %0\n\t" \
77 "csrw pmpcfg"#g", %1\n\t" \
78 "sfence.vma\n\t"\
79 ".align 2\n\t" \
80 "1: csrw mtvec, t0 \n\t" \
81 : : "r" (addr), "r" (pmpc) : "t0"); \
82}
