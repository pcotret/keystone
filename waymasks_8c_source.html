84// Simplest possible way choosing, with reserved way for each hart
85// Returns number of ways it couldn't allocate
86int _wm_choose_ways_for_hart(size_t n_ways, waymask_t* mask, unsigned int target_hart){
87
88 *mask = 0;
89 // Best effort, we may not be able to allocate n_ways
90 if(n_ways == 0)
91 return 0;
92
93 // Always allocate our 'reserved' way if available
94 if(! IS_WAY_ALLOCATED( GET_HART_WAY(target_hart) )){
95 *mask |= (0x1 << GET_HART_WAY(target_hart));
96 n_ways--;
97 }
98
99 // Find ways to allocate as available, ignoring reserved hart ways
100 int way = WM_MIN_FREE_WAY;
101 while(n_ways > 0 && way <= WM_MAX_FREE_WAY){
102 if( !IS_WAY_ALLOCATED(way) ){
103 *mask |= (0x1 << way);
104 n_ways--;
105 }
106 way++;
107 }
108
109 // Return number of UNallocated ways
110 return n_ways;
111}

114int _wm_lockout_ways(waymask_t mask, unsigned int master){
115
116 if(master > WM_NUM_MASTERS){

123 waymask_t* master_mask = WM_REG_ADDR(master);
124 waymask_t current_mask = *master_mask;
