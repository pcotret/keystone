21uintptr_t
22paging_alloc_backing_page() {
23 uintptr_t offs_update =
24 (paging_next_backing_page_offset + paging_inc_backing_page_offset_by) %
25 paging_backing_region_size();
26
27 /* no backing page available */
28 if (offs_update == 0) {
29 // cycled through all the pages
30 warn("no backing page avaiable");
31 return 0;
32 }
33
34 uintptr_t next_page =
35 paging_backing_region() + paging_next_backing_page_offset;
36 assert(IS_ALIGNED(next_page, RISCV_PAGE_BITS));
37
38 paging_next_backing_page_offset = offs_update;
39 return next_page;
40}

48static uintptr_t
49gcd(uintptr_t a, uintptr_t b) {
50 while (b) {
51 uintptr_t tmp = b;
52 b = a % b;
53 a = tmp;
54 }
55 return a;
56}

58static uintptr_t
59find_coprime_of(uintptr_t n) {
60 uintptr_t res;
61 do {
62 res = n / 2 + sbi_random() % (n / 2);
63 } while (gcd(res, n) != 1);
64 return res;
65}

78static uint64_t*
79pswap_pageout_ctr(uintptr_t page) {
80 assert(paging_backpage_inbounds(page));
81 size_t idx = (page - paging_backing_region()) >> RISCV_PAGE_BITS;
82 size_t indirect_idx = idx / (RISCV_PAGE_SIZE / 8);
83 size_t interior_idx = idx % (RISCV_PAGE_SIZE / 8);
84 assert(indirect_idx < NUM_CTR_INDIRECTS);
85
86 if (!ctr_indirect_ptrs[indirect_idx]) {
87 ctr_indirect_ptrs[indirect_idx] = paging_alloc_backing_page();
88 // Fill ptr pages with random values so our counters start unpredictable
89 rt_util_getrandom((void*)ctr_indirect_ptrs[indirect_idx], RISCV_PAGE_SIZE);
90 }
91
92 return (uint64_t*)(ctr_indirect_ptrs[indirect_idx]) + interior_idx;
93}

100static void
101pswap_establish_boot_key(void) {
102 uint8_t boot_key_tmp[32];
103
104 if (atomic_load(&pswap_boot_key_set)) {
105 // Key already set
106 return;
107 }
108
109 rt_util_getrandom(boot_key_tmp, 32);
110
111 if (atomic_flag_test_and_set(&pswap_boot_key_reserved)) {
112 // Lost the race; key already being set. Spin until finished.
113 while (!atomic_load(&pswap_boot_key_set))
114 ;
115 return;
116 }
117
118 memcpy(pswap_boot_key, boot_key_tmp, 32);
119 atomic_store(&pswap_boot_key_set, true);
120}

175/* evict a page from EPM and store it to the backing storage
176 * back_page (PA1) <-- epm_page (PA2) <-- swap_page (PA1)
177 * if swap_page is 0, no need to write epm_page
178 */
179void
180page_swap_epm(uintptr_t back_page, uintptr_t epm_page, uintptr_t swap_page) {
181 assert(paging_epm_inbounds(epm_page));
182 assert(paging_backpage_inbounds(back_page));
183
184 char buffer[RISCV_PAGE_SIZE] = {};
185 if (swap_page) {
186 assert(swap_page == back_page);
187 memcpy(buffer, (void*)swap_page, RISCV_PAGE_SIZE);
188 }
189
190 uint64_t* pageout_ctr = pswap_pageout_ctr(back_page);
191 uint64_t old_pageout_ctr = *pageout_ctr;
192 uint64_t new_pageout_ctr = old_pageout_ctr + 1;
193
194 uint8_t new_hash[32];
195 pswap_hash(new_hash, (void*)epm_page, new_pageout_ctr);
196 pswap_encrypt((void*)epm_page, (void*)back_page, new_pageout_ctr);
197
198 if (swap_page) {
199 uint8_t old_hash[32];
200 pswap_decrypt((void*)buffer, (void*)epm_page, old_pageout_ctr);
201 pswap_hash(old_hash, (void*)epm_page, old_pageout_ctr);
202
203#ifdef USE_PAGE_HASH
204 bool ok = merk_verify(&paging_merk_root, back_page, old_hash);
205 assert(ok);
206#endif
207 }
208
209#ifdef USE_PAGE_HASH
210 merk_insert(&paging_merk_root, back_page, new_hash);
211#endif
212
213 *pageout_ctr = new_pageout_ctr;
214
215 return;
216}
